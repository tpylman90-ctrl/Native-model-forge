cmake_minimum_required(VERSION 3.22)
project(model_forge_native_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(forge_geometry STATIC geometry.cpp)
add_library(forge_document STATIC document.cpp)
target_include_directories(forge_document PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(forge_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(ANDROID)
    add_library(model_forge_core SHARED jni_bridge.cpp)
    target_link_libraries(model_forge_core PRIVATE forge_geometry forge_document)
else()
    enable_testing()
    add_executable(geometry_test tests/geometry_test.cpp)
    target_link_libraries(geometry_test PRIVATE forge_geometry)
    add_test(NAME native_geometry_box COMMAND geometry_test)
    add_executable(document_test tests/document_test.cpp)
    target_link_libraries(document_test PRIVATE forge_document)
    add_test(NAME hybrid_document_transactions COMMAND document_test)
endif()
